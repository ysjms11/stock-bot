def main():
    # ── 1) regime.py 치환 ──
    src = REG.read_text(encoding="utf-8")
    if "E 하이브리드 v2" in src:
        print("[skip] regime.py 이미 v2 적용됨")
    elif OLD not in src:
        sys.exit("[abort] regime.py에서 기대한 v1 블록을 찾지 못함 — 코드가 변경된 듯. "
                 "data/FIX_regime_meltup_20260717.md 보고 수동 적용 필요. (아무것도 변경 안 함)")
    else:
        bak = str(REG) + ".bak_20260717"
        shutil.copy2(REG, bak)
        REG.write_text(src.replace(OLD, NEW, 1), encoding="utf-8")
        try:
            py_compile.compile(str(REG), doraise=True)
        except Exception as e:
            shutil.copy2(bak, REG)
            sys.exit(f"[abort] 컴파일 실패 → 백업 복원함: {e}")
        print(f"[ok] regime.py v2 적용 + 컴파일 확인 (백업: {bak})")

    # ── 2) test_regime.py 테스트 추가 ──
    t = TST.read_text(encoding="utf-8")
    if TEST_MARK in t:
        print("[skip] test_regime.py 이미 테스트 있음")
    else:
        anchor = 'if __name__ == "__main__":'
        if anchor not in t:
            sys.exit("[abort] test_regime.py 앵커를 찾지 못함 — 테스트 수동 추가 필요 "
                     "(data/FIX_regime_meltup_20260717.md §4)")
        TST.write_text(t.replace(anchor, TEST_CODE + "\n\n" + anchor, 1), encoding="utf-8")
        py_compile.compile(str(TST), doraise=True)
        print("[ok] test_regime.py 방향 게이트 테스트 5케이스 추가")

    # ── 3) pytest ──
    r = subprocess.run([sys.executable, "-m", "pytest", "test_regime.py", "-q"], cwd=str(ROOT))
    if r.returncode == 0:
        print("\n[done] 전체 green ✅ — 봇 재시작하면 적용 완료.")
        print("       재시작 후 get_regime 호출 → KR 로직에 '상방 멜트업 과열' 문구 확인.")
        print("       커밋: git add kis_api/regime.py test_regime.py && git commit")
    else:
        print(f"\n[warn] pytest exit={r.returncode} — 실패 테스트 확인 필요. 원복은 백업 파일로 가능.")
        sys.exit(r.returncode)


main()
